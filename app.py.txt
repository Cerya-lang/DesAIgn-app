import streamlit as st
import streamlit.components.v1 as components
import os
import json
import requests
import google.generativeai as genai
from google.oauth2 import service_account

# --- CONFIGURATION ---
st.set_page_config(page_title="DesAIgn Studio", page_icon="🎨", layout="wide")

HF_TOKEN = os.environ.get('HF_TOKEN')
GOOGLE_JSON = os.environ.get('GOOGLE_APPLICATION_CREDENTIALS_JSON')

# --- INITIALISATION GEMINI ---
try:
    if GOOGLE_JSON:
        info = json.loads(GOOGLE_JSON)
        cred = service_account.Credentials.from_service_account_info(info, scopes=['https://www.googleapis.com/auth/cloud-platform'])
        genai.configure(credentials=cred)
        model = genai.GenerativeModel('gemini-1.5-flash')
except Exception as e:
    st.error(f"Erreur Configuration Google : {e}")

    </style>
    """, unsafe_allow_html=True)

# INTERFACE PRINCIPALE
# ==========================================
st.title("🎨 DesAIgn | Studio Intelligent")
st.caption("Plateforme collaborative de design industriel et mode")

tabs = st.tabs(["💬 Recherche & Ingénierie (IA)", "🧊 Moteur de Rendu 3D", "📞 War Room"])

# --- ONGLET 1 : CHAT ET ANALYSE PROFONDE ---
with tabs[0]:
    st.info("💡 **Astuce :** Demande à l'IA d'analyser une marque (ex: Nike, JDM) et de te créer le 'Prompt 3D' parfait à utiliser dans l'onglet suivant.")
    
    if "messages" not in st.session_state: st.session_state.messages = []
    
    chat_box = st.container(height=400)
    for msg in st.session_state.messages:
        with chat_box.chat_message(msg["role"]): st.markdown(msg["content"])
    
    if prompt := st.chat_input("Fais une recherche sur un design, une voiture ou un vêtement..."):
        st.session_state.messages.append({"role": "user", "content": prompt})
        with chat_box.chat_message("user"): st.markdown(prompt)
        
        with chat_box.chat_message("assistant"):
            with st.spinner("DesAIgn effectue une recherche dans sa base de données web..."):
                try:
                    # PROMPT SYSTÈME MODIFIÉ POUR LA RECHERCHE
                    full_prompt = f"""Tu es l'IA de DesAIgn. L'utilisateur te fait cette demande : "{prompt}".
                    Agis comme un moteur de recherche et un ingénieur en design. 
                    1. Fais une analyse détaillée du sujet (contexte culturel, esthétique, spécifications techniques).
                    2. SI l'utilisateur parle de modélisation 3D, explique-lui que tu ne peux pas générer la 3D ici, mais fournis-lui un "PROMPT 3D EN ANGLAIS" très précis (optimisé pour Shap-E ou Meshy) qu'il pourra copier-coller dans l'onglet Moteur 3D."""
                    
                    response = model.generate_content(full_prompt)
                    st.markdown(response.text)
                    st.session_state.messages.append({"role": "assistant", "content": response.text})
                except Exception as e:
                    st.error(f"Erreur Gemini : {e}")

# --- ONGLET 2 : MOTEUR 3D ---
with tabs[1]:
    st.subheader("Laboratoire de Modélisation 3D")
    st.write("Colle ici le prompt en anglais généré par l'IA dans l'onglet Recherche.")
    
    prompt_3d = st.text_input("Prompt de modélisation :", "A futuristic cyberpunk mask with glowing details")
    
    if st.button("Lancer la fabrication (Shap-E)"):
        if not HF_TOKEN:
            st.warning("HF_TOKEN non configuré.")
        else:
            with st.spinner("Fabrication des polygones en cours..."):
                API_URL = "https://api-inference.huggingface.co/models/openai/shap-e"
                headers = {"Authorization": f"Bearer {HF_TOKEN}"}
                try:
                    r = requests.post(API_URL, headers=headers, json={"inputs": prompt_3d}, timeout=60)
                    if r.status_code == 200:
                        st.success("Modèle généré ! (Note: Limite de détails atteinte avec le forfait gratuit)")
                    else:
                        st.error(f"Moteur surchargé. Code: {r.status_code}.")
                except Exception as e:
                    st.error(f"Erreur : {e}")
